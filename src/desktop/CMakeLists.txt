find_package(OpenGL REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
find_package(SDL2_image CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(GLEW REQUIRED)
find_package(PNG REQUIRED)
find_package(Freetype REQUIRED)
find_package(Boost 1.69 REQUIRED COMPONENTS filesystem)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8)

set(vendor "${PROJECT_SOURCE_DIR}/third_party")
set(gource "${vendor}/Gource/src")
set(logstalgia "${vendor}/Logstalgia/src")
set(imgui "${vendor}/imgui")
set(imkit "${vendor}/imkit")
add_library(viz_imgui STATIC
  ${imgui}/imgui.cpp ${imgui}/imgui_draw.cpp ${imgui}/imgui_widgets.cpp ${imgui}/imgui_tables.cpp
  ${imgui}/misc/cpp/imgui_stdlib.cpp ${imgui}/backends/imgui_impl_sdl2.cpp ${imgui}/backends/imgui_impl_opengl2.cpp
  ${imkit}/src/theme.cpp ${imkit}/src/widgets.cpp ${imkit}/src/decor.cpp ${imkit}/src/accessibility.cpp
  ${imkit}/src/components.cpp ${imkit}/src/patterns.cpp ${imkit}/src/icons.cpp)
target_compile_features(viz_imgui PUBLIC cxx_std_20)
target_include_directories(viz_imgui PUBLIC ${imgui} ${imkit}/include)
target_link_libraries(viz_imgui PUBLIC SDL2::SDL2 OpenGL::GL)

# The pinned Core headers match. Use Gource's Core implementation for its
# compatibility profile and indexed texture handling.
file(GLOB common_sources "${gource}/core/*.cpp")
list(REMOVE_ITEM common_sources "${gource}/core/tga.cpp" "${gource}/core/timer.cpp")
add_library(viz_graphics STATIC ${common_sources})
target_compile_features(viz_graphics PRIVATE cxx_std_17)
target_compile_definitions(viz_graphics PUBLIC SDL_MAIN_HANDLED)
target_include_directories(viz_graphics PUBLIC ${gource}/core)
target_link_libraries(viz_graphics PUBLIC SDL2::SDL2 SDL2_image::SDL2_image GLEW::GLEW
  OpenGL::GL OpenGL::GLU glm::glm PNG::PNG Freetype::Freetype PkgConfig::PCRE2 Boost::filesystem)
if(WIN32)
  target_link_libraries(viz_graphics PUBLIC comdlg32)
elseif(APPLE)
  target_link_libraries(viz_graphics PUBLIC "-framework CoreFoundation")
endif()

file(GLOB gource_sources "${gource}/*.cpp" "${gource}/formats/*.cpp" "${gource}/tinyxml/*.cpp")
list(REMOVE_ITEM gource_sources "${gource}/main.cpp")
file(GLOB log_sources "${logstalgia}/*.cpp")
list(REMOVE_ITEM log_sources "${logstalgia}/main.cpp" "${logstalgia}/tests.cpp")
add_library(viz_gource STATIC gource_view.cpp ${gource_sources})
add_library(viz_logstalgia STATIC logstalgia_view.cpp ${log_sources})
target_include_directories(viz_gource PRIVATE "${gource}")
target_include_directories(viz_logstalgia PRIVATE "${logstalgia}")
target_compile_definitions(viz_logstalgia PRIVATE PositionSlider=LogstalgiaPositionSlider)
foreach(engine viz_gource viz_logstalgia)
  target_link_libraries(${engine} PUBLIC viz_graphics)
  target_compile_features(${engine} PRIVATE cxx_std_17)
endforeach()
if(UNIX)
  target_compile_definitions(viz_gource PRIVATE HAVE_TIMEGM HAVE_UNSETENV)
endif()

add_executable(bitcoin-viz-desktop desktop.cpp panel.cpp session.cpp)
target_link_libraries(bitcoin-viz-desktop PRIVATE viz_gource viz_logstalgia viz_imgui bitcoin_viz_core)
set_target_properties(bitcoin-viz-desktop PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(renderer Gource Logstalgia)
  set(assets "${vendor}/${renderer}/data")
  add_custom_command(TARGET bitcoin-viz-desktop POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${assets}" "$<TARGET_FILE_DIR:bitcoin-viz-desktop>/viz-data/${renderer}")
  install(DIRECTORY "${assets}/" DESTINATION "${CMAKE_INSTALL_BINDIR}/viz-data/${renderer}")
endforeach()
install(TARGETS bitcoin-viz-desktop RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${vendor}/README.md" DESTINATION "${CMAKE_INSTALL_DATADIR}/bitcoin-viz"
  RENAME third-party.md)
foreach(lib Gource Logstalgia)
  install(FILES "${vendor}/${lib}/COPYING" DESTINATION "${CMAKE_INSTALL_DATADIR}/bitcoin-viz/${lib}")
endforeach()
install(FILES "${imgui}/LICENSE.txt" DESTINATION "${CMAKE_INSTALL_DATADIR}/bitcoin-viz/imgui")
install(FILES "${imkit}/LICENSE" "${imkit}/THIRD_PARTY_NOTICES.md" DESTINATION "${CMAKE_INSTALL_DATADIR}/bitcoin-viz/imkit")
