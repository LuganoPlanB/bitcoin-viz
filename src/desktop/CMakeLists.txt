# The pinned Core headers match. Use Gource's Core implementation for its
# compatibility profile and indexed texture handling.
file(GLOB common_sources "${gource}/core/*.cpp")
list(REMOVE_ITEM common_sources "${gource}/core/tga.cpp" "${gource}/core/timer.cpp")
add_library(viz_graphics STATIC ${common_sources})
target_compile_features(viz_graphics PRIVATE cxx_std_17)
target_compile_definitions(viz_graphics PUBLIC SDL_MAIN_HANDLED)
target_include_directories(viz_graphics PUBLIC ${gource}/core)
target_link_libraries(viz_graphics PUBLIC ${sdl_target} ${sdl_image_target} GLEW::GLEW
  OpenGL::GL OpenGL::GLU glm::glm PNG::PNG Freetype::Freetype ${pcre2_target} Boost::filesystem)
if(WIN32)
  target_compile_definitions(viz_graphics PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(viz_graphics PUBLIC comdlg32)
elseif(APPLE)
  target_link_libraries(viz_graphics PUBLIC "-framework CoreFoundation")
endif()

file(GLOB gource_sources "${gource}/*.cpp" "${gource}/formats/*.cpp" "${gource}/tinyxml/*.cpp")
list(REMOVE_ITEM gource_sources "${gource}/main.cpp")
file(GLOB log_sources "${logstalgia}/*.cpp")
list(REMOVE_ITEM log_sources "${logstalgia}/main.cpp" "${logstalgia}/tests.cpp")

  target_compile_features(${engine} PRIVATE cxx_std_17)

if(BUILD_TESTING)
  add_executable(bitcoin-viz-scene-test "${PROJECT_SOURCE_DIR}/tests/desktop_scene.cpp")
  target_include_directories(bitcoin-viz-scene-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${vendor}")
  target_link_libraries(bitcoin-viz-scene-test PRIVATE viz_gource viz_logstalgia bitcoin_viz_core)
  add_test(NAME desktop-scene-projections COMMAND bitcoin-viz-scene-test projections)
  add_test(NAME desktop-scene-validation COMMAND bitcoin-viz-scene-test validation)
  set_tests_properties(desktop-scene-projections desktop-scene-validation PROPERTIES TIMEOUT 20)
  set(BITCOIN_VIZ_TEST_CORE_BIN_DIR "" CACHE PATH "Directory containing bitcoind and bitcoin-cli for the desktop regtest test")
  if(BITCOIN_VIZ_TEST_CORE_BIN_DIR)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
  else()
    find_package(Python3 QUIET COMPONENTS Interpreter)
  endif()
  if(Python3_Interpreter_FOUND)
    add_executable(bitcoin-viz-session-test "${PROJECT_SOURCE_DIR}/tests/desktop_session.cpp" session.cpp)
    target_include_directories(bitcoin-viz-session-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(bitcoin-viz-session-test PRIVATE bitcoin_viz_core)
    foreach(case paths discovery switch sequence)
      add_test(NAME desktop-node-${case} COMMAND "${Python3_EXECUTABLE}" -u -B
        "${PROJECT_SOURCE_DIR}/tests/desktop_session.py" ${case} "$<TARGET_FILE:bitcoin-viz-session-test>")
      set_tests_properties(desktop-node-${case} PROPERTIES TIMEOUT 35)
    endforeach()
    if(BITCOIN_VIZ_TEST_CORE_BIN_DIR)
      set(viz_test_bitcoind "${BITCOIN_VIZ_TEST_CORE_BIN_DIR}/bitcoind${CMAKE_EXECUTABLE_SUFFIX}")
      set(viz_test_bitcoin_cli "${BITCOIN_VIZ_TEST_CORE_BIN_DIR}/bitcoin-cli${CMAKE_EXECUTABLE_SUFFIX}")
      if(NOT EXISTS "${viz_test_bitcoind}" OR NOT EXISTS "${viz_test_bitcoin_cli}")
        message(FATAL_ERROR "BITCOIN_VIZ_TEST_CORE_BIN_DIR must contain bitcoind and bitcoin-cli")
      endif()
      add_test(NAME desktop-node-regtest COMMAND "${Python3_EXECUTABLE}" -u -B
        "${PROJECT_SOURCE_DIR}/tests/desktop_regtest.py" "$<TARGET_FILE:bitcoin-viz-session-test>"
        "$<TARGET_FILE:bitcoin-viz>" "${viz_test_bitcoind}" "${viz_test_bitcoin_cli}")
      set_tests_properties(desktop-node-regtest PROPERTIES TIMEOUT 85)
    endif()
  else()
    message(STATUS "Python 3 unavailable; desktop Session tests are disabled")
  endif()
endif()

foreach(renderer Gource Logstalgia)
  set(assets "${vendor}/${renderer}/data")
  set(staged_assets "${CMAKE_CURRENT_BINARY_DIR}/assets/${renderer}")
  file(GLOB_RECURSE runtime_files RELATIVE "${assets}" CONFIGURE_DEPENDS
    "${assets}/*.png" "${assets}/*.tga" "${assets}/*.ttf" "${assets}/*.frag" "${assets}/*.vert" "${assets}/fonts/README")
  foreach(asset IN LISTS runtime_files)
    configure_file("${assets}/${asset}" "${staged_assets}/${asset}" COPYONLY)
  endforeach()
  add_custom_command(TARGET bitcoin-viz-desktop POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${staged_assets}" "${build_assets}/${renderer}" VERBATIM)
  install(DIRECTORY "${staged_assets}/" DESTINATION "${install_assets}/${renderer}")
endforeach()
