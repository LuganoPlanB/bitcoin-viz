cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" BITCOIN_VIZ_SOURCE_VERSION LIMIT_COUNT 1)
if(NOT DEFINED BITCOIN_VIZ_VERSION)
  set(BITCOIN_VIZ_VERSION "${BITCOIN_VIZ_SOURCE_VERSION}")
endif()
if(NOT BITCOIN_VIZ_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "BITCOIN_VIZ_VERSION must be a strict major.minor.patch version")
endif()
project(bitcoin-viz VERSION ${BITCOIN_VIZ_VERSION} LANGUAGES CXX)
configure_file(src/version.hpp.in generated/bitcoin_viz_version.hpp @ONLY)

include(CTest)
include(GNUInstallDirs)
option(BITCOIN_VIZ_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
find_program(BITCOIN_VIZ_CCACHE ccache)
if(BITCOIN_VIZ_CCACHE AND NOT DEFINED CMAKE_CXX_COMPILER_LAUNCHER)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${BITCOIN_VIZ_CCACHE}" CACHE STRING "Compiler launcher")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
find_package(SQLite3 3.24 REQUIRED)
find_package(CURL 7.68 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(ZeroMQ CONFIG QUIET)
if(TARGET libzmq-static)
  set(BITCOIN_VIZ_ZMQ_TARGET libzmq-static)
elseif(TARGET libzmq)
  set(BITCOIN_VIZ_ZMQ_TARGET libzmq)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ZMQ REQUIRED IMPORTED_TARGET libzmq>=4.3)
  set(BITCOIN_VIZ_ZMQ_TARGET PkgConfig::ZMQ)
endif()

add_library(bitcoin_viz_core STATIC
  src/storage.cpp src/model.cpp src/core.cpp src/observer.cpp
  src/render.cpp src/logstalgia.cpp src/capture.cpp src/platform.cpp)
target_compile_features(bitcoin_viz_core PUBLIC cxx_std_17)
set_target_properties(bitcoin_viz_core PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(bitcoin_viz_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(bitcoin_viz_core SYSTEM PUBLIC third_party)
target_link_libraries(bitcoin_viz_core PUBLIC SQLite::SQLite3 CURL::libcurl
  OpenSSL::Crypto ${BITCOIN_VIZ_ZMQ_TARGET} Threads::Threads)
if(WIN32)
  target_compile_definitions(bitcoin_viz_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
target_compile_options(bitcoin_viz_core PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)

add_executable(bitcoin-viz src/main.cpp)
set_target_properties(bitcoin-viz PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(bitcoin-viz PRIVATE bitcoin_viz_core)
target_compile_options(bitcoin-viz PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
if(BITCOIN_VIZ_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "BITCOIN_VIZ_SANITIZE requires GCC or Clang")
  endif()
  target_compile_options(bitcoin_viz_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bitcoin_viz_core PUBLIC -fsanitize=address,undefined)
endif()
install(TARGETS bitcoin-viz RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/bitcoin-viz)
install(FILES third_party/nlohmann/LICENSE.MIT
  DESTINATION ${CMAKE_INSTALL_DATADIR}/bitcoin-viz RENAME nlohmann-LICENSE.MIT)

if(BUILD_TESTING)
  add_executable(bitcoin-viz-contract tests/contract.cpp)
  set_target_properties(bitcoin-viz-contract PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_link_libraries(bitcoin-viz-contract PRIVATE bitcoin_viz_core)
  add_test(NAME native-contract COMMAND bitcoin-viz-contract)
  set_tests_properties(native-contract PROPERTIES TIMEOUT 30)
  add_test(NAME cli COMMAND ${CMAKE_COMMAND}
    "-DBINARY=$<TARGET_FILE:bitcoin-viz>" "-DVERSION=${PROJECT_VERSION}"
    "-DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/cli-test" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  add_test(NAME release-tools COMMAND ${CMAKE_COMMAND}
    "-DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/release-test" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/release.cmake")
  set_tests_properties(cli release-tools PROPERTIES TIMEOUT 60)
endif()
